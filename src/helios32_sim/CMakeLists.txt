cmake_minimum_required(VERSION 3.5)
project(helios32_sim)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
foreach(dep gazebo_dev gazebo_ros rclcpp sensor_msgs std_msgs nav_msgs geometry_msgs gazebo_msgs tf2 tf2_ros tf2_geometry_msgs pcl_ros pcl_conversions message_filters)
  find_package(${dep} REQUIRED)
endforeach()
add_library(helios32_ray SHARED src/helios32_ray.cpp)
ament_target_dependencies(helios32_ray gazebo_dev gazebo_ros rclcpp sensor_msgs std_msgs)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/vehicleSimulator.cpp")
  find_package(OpenCV REQUIRED)
  find_package(PCL REQUIRED)
  add_executable(vehicleSimulator src/vehicleSimulator.cpp)
  ament_target_dependencies(vehicleSimulator rclcpp sensor_msgs std_msgs nav_msgs geometry_msgs gazebo_msgs gazebo_ros tf2 tf2_ros tf2_geometry_msgs pcl_ros pcl_conversions message_filters)
  target_link_libraries(vehicleSimulator ${OpenCV_LIBRARIES} ${PCL_LIBRARIES})
  install(TARGETS vehicleSimulator DESTINATION lib/${PROJECT_NAME})
endif()
install(TARGETS helios32_ray LIBRARY DESTINATION lib)
ament_package()
